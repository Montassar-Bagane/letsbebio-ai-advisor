%------------------------------------------------ figure placeholder macro --
% If graphic file exists in figures/, include it; otherwise render an elegant placeholder box.
% Ensures compilation never crashes on missing images and registers in \listoffigures.
\newcommand{\placeholderscreenshot}[4][0.9\textwidth]{%
  \begin{figure}[H]
    \centering
    \IfFileExists{figures/#2}{%
      \includegraphics[width=#1]{figures/#2}%
    }{%
      \setlength{\fboxsep}{10pt}%
      \fbox{%
        \begin{minipage}[c][4.6cm][c]{#1}
          \centering
          {\color{epired}\large\bfseries [ SCREENSHOT / FIGURE PLACEHOLDER ]}\\[7pt]
          {\small\bfseries #3}\\[6pt]
          {\footnotesize\color{rule}\texttt{Target file: figures/#2}}\\[4pt]
          {\scriptsize\itshape\color{rule}(Drop your screenshot or diagram into the \texttt{figures/} folder)}
        \end{minipage}%
      }%
    }%
    \caption{#3}
    \label{#4}
  \end{figure}
}
